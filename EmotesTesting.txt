-- EMOTE STATE & EXPRESSION HANDLER
local function setEmotingState(character, state)
    if not character then return end
    
    -- Set the requested IsEmoting attribute
    character:SetAttribute("IsEmoting", state)
    
    local head = character:FindFirstChild("Head")
    if head then
        local face = head:FindFirstChild("face")
        if face and face:IsA("Decal") then
            if state then
                -- Save original face if not already saved, then switch to Happy face
                if not head:FindFirstChild("OriginalFace") then
                    local orig = Instance.new("StringValue")
                    orig.Name = "OriginalFace"
                    orig.Value = face.Texture
                    orig.Parent = head
                end
                face.Texture = "rbxassetid://94346747799192" -- Classic Happy face ID
            else
                -- Restore original face when emote stops
                local orig = head:FindFirstChild("OriginalFace")
                if orig then
                    face.Texture = orig.Value
                    orig:Destroy()
                end
            end
        end
    end
end

-- CAMERA
local function enableCamLock(character)
    local head = character:FindFirstChild("Head")
    if head then
        camera.CameraSubject = head
        camera.CameraType = Enum.CameraType.Custom
    end
end

local function disableCamLock(character)
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if humanoid then
        camera.CameraSubject = humanoid
        camera.CameraType = Enum.CameraType.Custom
    end
end

-- HELPER TO CLEAN PROPS SAFELY
local function clearCurrentProps()
    if cloneTask then
        task.cancel(cloneTask)
        cloneTask = nil
    end

    if type(currentProp) == "table" then
        for _, prop in ipairs(currentProp) do
            if typeof(prop) == "Instance" then
                prop:Destroy()
            end
        end
    elseif typeof(currentProp) == "Instance" then
        currentProp:Destroy()
    end
    currentProp = {}
end

local function stopAll()
    actionId += 1
    
    clearCurrentProps()

    local character = player.Character
    if character then
        setEmotingState(character, false) -- Disable IsEmoting attribute and reset expression
        
        local oldMewProp = character:FindFirstChild("MEW_HANDLE")
        if oldMewProp then
            oldMewProp:Destroy()
        end
        local oldCloneHolder = character:FindFirstChild("c00lholder")
        if oldCloneHolder then
            oldCloneHolder:Destroy()
        end
        local oldAura = character:FindFirstChild("AuraMonster")
        if oldAura then
            oldAura:Destroy()
        end

        local oldlxt = character:FindFirstChild("LRXGTR")
        if oldlxt then
            oldlxt:Destroy()
        end
    end

    if currentAnim then
        currentAnim:Stop()
        currentAnim = nil
    end

    if currentSound then
        currentSound:Stop()
        currentSound:Destroy()
        currentSound = nil
    end

    if loopConnection then
        loopConnection:Disconnect()
        loopConnection = nil
    end

    if autoWalkConnection then
        autoWalkConnection:Disconnect()
        autoWalkConnection = nil
    end

    if speedTrackConnection then
        speedTrackConnection:Disconnect()
        speedTrackConnection = nil
    end

    if character then
        disableCamLock(character)
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            humanoid.WalkSpeed = 16
        end
    end
end

-- SPEED TRACKER
local function trackAnimationSpeed(track, humanoid, targetSpeed)
    if speedTrackConnection then speedTrackConnection:Disconnect() end
    speedTrackConnection = RunService.RenderStepped:Connect(function()
        if track and track.IsPlaying and humanoid then
            humanoid.WalkSpeed = targetSpeed
        else
            if speedTrackConnection then
                speedTrackConnection:Disconnect()
                speedTrackConnection = nil
            end
            if humanoid then humanoid.WalkSpeed = 16 end
        end
    end)
end

local function StartLoop(introAnimId, introSoundId, transitionDelay, loopAnimId, loopSoundId, isSpeedEmote, speedVal, propFunction)
    stopAll()
    local myActionId = actionId
    local character = getCharacter()
    
    setEmotingState(character, true) -- Enable IsEmoting attribute and set expression to Happy
    enableCamLock(character)
    
    -- Spawn the prop here so it survives stopAll() and initializes with the emote
    if propFunction and type(propFunction) == "function" then
        currentProp = propFunction(character)
    end
    
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    local animator = humanoid:FindFirstChildOfClass("Animator") or Instance.new("Animator", humanoid)

    -- Play Intro Animation
    local introAnim = Instance.new("Animation")
    introAnim.AnimationId = introAnimId
    local introTrack = animator:LoadAnimation(introAnim)
    introTrack.Priority = Enum.AnimationPriority.Action4
    introTrack.Looped = false
    introTrack:Play()
    currentAnim = introTrack

    -- Play Intro Sound
    if introSoundId then
        local sound = Instance.new("Sound")
        sound.SoundId = introSoundId
        sound.Volume = globalVolume
        sound.Parent = character:WaitForChild("HumanoidRootPart")
        sound:Play()
        currentSound = sound
    end

    if isSpeedEmote and speedVal then
        trackAnimationSpeed(introTrack, humanoid, speedVal)
    end

    -- Transition to Loop Stage
    task.delay(transitionDelay, function()
        if actionId ~= myActionId then return end

        if currentAnim then currentAnim:Stop() end
        if currentSound then currentSound:Stop() currentSound:Destroy() end

        local loopAnim = Instance.new("Animation")
        loopAnim.AnimationId = loopAnimId
        local loopTrack = animator:LoadAnimation(loopAnim)
        loopTrack.Priority = Enum.AnimationPriority.Action4
        loopTrack.Looped = true
        loopTrack:Play()
        currentAnim = loopTrack

        if loopSoundId then
            local sound = Instance.new("Sound")
            sound.SoundId = loopSoundId
            sound.Looped = true
            sound.Volume = globalVolume
            sound.Parent = character:WaitForChild("HumanoidRootPart")
            sound:Play()
            currentSound = sound
        end

        if isSpeedEmote and speedVal then
            trackAnimationSpeed(loopTrack, humanoid, speedVal)
        end
    end)
end

local function LoadSukunaAssets(character)
    local emotesFolder = ReplicatedStorage:FindFirstChild("Assets") 
        and ReplicatedStorage.Assets:FindFirstChild("Emotes") 
        and ReplicatedStorage.Assets.Emotes:FindFirstChild("Sukuna")

    if not emotesFolder then 
        warn("[Sukuna] Could not locate Assets.Emotes.Sukuna in ReplicatedStorage!")
        return {} 
    end

    local assets = {
        emotesFolder:FindFirstChild("DismantleFX"),
        emotesFolder:FindFirstChild("LeftHandle"),
        emotesFolder:FindFirstChild("RightHandle"),
    }

    local spawnedAssets = {}

    for _, raw in ipairs(assets) do
        if raw then
            local clone = raw:Clone()
            clone.Parent = character

            if raw.Name == "LeftHandle" or raw.Name == "RightHandle" then
                local side = (raw.Name == "LeftHandle") and "Left" or "Right"
                local arm = character:FindFirstChild(side .. " Arm") or character:FindFirstChild(side .. "Hand")

                if arm then
                    for _, child in ipairs(clone:GetDescendants()) do
                        if child:IsA("BasePart") then
                            child.Color = arm.Color
                            child.CanCollide = false
                            child.Anchored = false
                        end
                    end

                    local motor = clone:FindFirstChildWhichIsA("Motor6D", true)
                    if motor then
                        motor.Part0 = arm
                        if not motor.Part1 then
                            motor.Part1 = clone:IsA("BasePart") and clone or clone:FindFirstChildWhichIsA("BasePart")
                        end
                    end
                end
            end

            table.insert(spawnedAssets, clone)
        end
    end

    return spawnedAssets
end

local function SpawnCkClones(character, animId)
    local emoteFolder = ReplicatedStorage:FindFirstChild("Assets") 
        and ReplicatedStorage.Assets:FindFirstChild("Emotes")
        and ReplicatedStorage.Assets.Emotes:FindFirstChild("#CkCloning")

    if not emoteFolder then return {} end

    local holder = emoteFolder:FindFirstChild("c00lholder") and emoteFolder.c00lholder:Clone() or Instance.new("Folder")
    holder.Name = "c00lholder"
    holder.Parent = character

    local startTime = tick()
    local animAsset = animId or "rbxassetid://119932181590245"

    cloneTask = task.spawn(function()
        for i = 1, 3 do
            task.wait(6)
            if not character.Parent or not holder.Parent then break end
            
            local cloneTemplate = emoteFolder:FindFirstChild("c00l")
            if cloneTemplate then
                local cloneRig = cloneTemplate:Clone()
                cloneRig.Name = "c00lCLONE"
                cloneRig.Parent = holder
                
                if cloneRig.PrimaryPart then cloneRig.PrimaryPart.Anchored = false end
                local torso = cloneRig:FindFirstChild("Torso") or cloneRig:FindFirstChild("UpperTorso")
                if torso then torso.CanCollide = false end

                local root = character.PrimaryPart or character:FindFirstChild("HumanoidRootPart")
                if root and cloneRig.PrimaryPart then
                    cloneRig:PivotTo(root.CFrame * CFrame.new(-(i + 0.25), 0, i + 0.25))
                    
                    local weld = cloneRig.PrimaryPart:FindFirstChildOfClass("WeldConstraint") or Instance.new("WeldConstraint")
                    weld.Part0 = cloneRig.PrimaryPart
                    weld.Part1 = root
                    weld.Parent = cloneRig.PrimaryPart
                end

                local hum = cloneRig:FindFirstChildOfClass("Humanoid")
                if hum then
                    local animator = hum:FindFirstChildOfClass("Animator") or Instance.new("Animator", hum)
                    local animObj = Instance.new("Animation")
                    animObj.AnimationId = animAsset
                    local track = animator:LoadAnimation(animObj)
                    track:Play(0)
                    track.TimePosition = tick() - startTime
                end
            end
        end
    end)

    return {holder}
end

        if swordMotor and swordMotor:IsA("Motor6D") then
            swordMotor.Part0 = rightArm
            swordMotor.Part1 = primary or swordMotor.Parent

        for _, desc in ipairs(swordClone:GetDescendants()) do

                desc.Anchored = false
                desc.CanCollide = false
            end

        table.insert(spawnedAssets, swordClone)

local function SpawnObservingEyes(character)
    local spawnedAssets = {}
    if not character then return spawnedAssets end

    local head = character:FindFirstChild("Head")
    if not head then return spawnedAssets end

    local observingFolder = ReplicatedStorage:FindFirstChild("Assets") 
        and ReplicatedStorage.Assets:FindFirstChild("Emotes") 
        and ReplicatedStorage.Assets.Emotes:FindFirstChild("observing")

    if not observingFolder then return spawnedAssets end

    local function attachEye(eyeTemplate, prefix)
        if not eyeTemplate then return end

        local eyeClone = eyeTemplate:Clone()
        eyeClone.Parent = character

        local mainPart = eyeClone:FindFirstChild(prefix .. ".1") 
            or eyeClone.PrimaryPart 
            or eyeClone:FindFirstChildWhichIsA("BasePart")

        if mainPart then
            mainPart.Anchored = false
            mainPart.CanCollide = false

            local motor = mainPart:FindFirstChild("Motor6D") or mainPart:FindFirstChildWhichIsA("Motor6D")
            if motor then
                motor.Part0 = head
                motor.Part1 = mainPart
            else
                local newMotor = Instance.new("Motor6D")
                newMotor.Name = "Motor6D"
                newMotor.Part0 = head
                newMotor.Part1 = mainPart
                newMotor.Parent = mainPart
            end

            for _, subPart in ipairs(eyeClone:GetDescendants()) do
                if subPart:IsA("BasePart") and subPart ~= mainPart then
                    subPart.Anchored = false
                    subPart.CanCollide = false

                    local partName = subPart.Name
                    local partNum = tonumber(partName:match("%d+$"))

                    if partNum and partNum > 1 then
                        local targetName = prefix .. "." .. (partNum + 1)
                        local targetPart = eyeClone:FindFirstChild(targetName, true)

                        local motor = subPart:FindFirstChild(targetName) or subPart:FindFirstChildWhichIsA("Motor6D")
                        if motor and targetPart then
                            motor.Part0 = subPart
                            motor.Part1 = targetPart
                        end
                    end
                end
            end

            local part4 = eyeClone:FindFirstChild(prefix .. ".4", true)
            local part3 = eyeClone:FindFirstChild(prefix .. ".3", true)

            if part4 then
                local subWeld = part4:FindFirstChild("EyeSubWeld") or part4:FindFirstChildWhichIsA("WeldConstraint")
                if not subWeld then
                    subWeld = Instance.new("WeldConstraint")
                    subWeld.Name = "EyeSubWeld"
                    subWeld.Parent = part4
                end

                subWeld.Part0 = part3 or mainPart
                subWeld.Part1 = part4
            end
        end

        table.insert(spawnedAssets, eyeClone)
    end

    attachEye(observingFolder:FindFirstChild("LeftEye"), "l")
    attachEye(observingFolder:FindFirstChild("RightEye"), "r")

    return spawnedAssets
end

local function SpawnMewEmote(character)
    local emoteTemplate = ReplicatedStorage:FindFirstChild("Assets") 
        and ReplicatedStorage.Assets:FindFirstChild("Emotes") 
        and ReplicatedStorage.Assets.Emotes:FindFirstChild("#MewMew") 
        and ReplicatedStorage.Assets.Emotes["#MewMew"]:FindFirstChild("MEWEMOTE")

    local rightArm = character:FindFirstChild("Right Arm") or character:FindFirstChild("RightHand")
    local leftArm = character:FindFirstChild("Left Arm") or character:FindFirstChild("LeftHand")
    local primaryPart = character.PrimaryPart or character:FindFirstChild("HumanoidRootPart")

    if not (rightArm and leftArm and primaryPart) then return {} end

    local clonedEmote = emoteTemplate:Clone()
    clonedEmote.Name = "MEW_HANDLE"

    local rightLight = clonedEmote:FindFirstChild("RightLight")
    if rightLight and rightLight:FindFirstChild("RIGHTWELD") then
        rightLight.RIGHTWELD.Part0 = rightArm
    end

    local leftLight = clonedEmote:FindFirstChild("LeftLight")
    if leftLight and leftLight:FindFirstChild("LEFTWELD") then
        leftLight.LEFTWELD.Part0 = leftArm
    end

    local nothingWeld = clonedEmote:FindFirstChild("NOTHINGWELD", true)
    if nothingWeld then
        nothingWeld.Part0 = primaryPart
    end

    clonedEmote.Parent = character
    return {clonedEmote}
end

end

local function SpecialTime(animTrack, startTime, endTime)
    if not animTrack then return end

    animTrack.TimePosition = startTime

    local connection
    connection = RunService.Heartbeat:Connect(function()
        if not animTrack.IsPlaying then
            connection:Disconnect()
            return
        end

        if animTrack.TimePosition >= endTime or animTrack.TimePosition < startTime then
            animTrack.TimePosition = startTime
        end
    end)
end

-- SLIDER DRAGGING SYSTEM
local isDragging = false

local function updateVolume(input)
    local trackAbsPos = sliderTrack.AbsolutePosition.X
    local trackAbsSize = sliderTrack.AbsoluteSize.X
    local mouseX = input.Position.X
    
    local relativeX = math.clamp(mouseX - trackAbsPos, 0, trackAbsSize)
    local percent = relativeX / trackAbsSize
    
    globalVolume = percent
    sliderFill.Size = UDim2.new(percent, 0, 1, 0)
    volTitle.Text = "Volume: " .. math.floor(percent * 100) .. "%"
    
    if currentSound then
        currentSound.Volume = globalVolume
    end
end

volPanel.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        isDragging = true
        updateVolume(input)
    end
end)

volPanel.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        isDragging = false
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        updateVolume(input)
    end
end)
